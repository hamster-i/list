# Имя конечного исполняемого файла
TARGET = program

# Компилятор и флаги компиляции
CC = gcc
CFLAGS = -Wall -Wextra -std=c11 -g -I./src -I./test

# Поиск всех исходных файлов .c в текущей папке, src и test
SRCS = main.c (wildcard src/*.c) (wildcard test/*.c)

# Преобразование списка файлов .c в список объектных файлов .o
OBJS = \$(SRCS:.c=.o)

# Главное правило (вызывается по умолчанию при запуске `make`)
all: \$(TARGET)

# Правило для сборки исполняемого файла из объектных
	\$(TARGET): \((OBJS)\)(CC) (OBJS) -o (TARGET)

# Правило для компиляции каждого файла .c в .o
%.o: %.c
	(CC) (CFLAGS) -c < -o @

# Очистка проекта от объектных файлов и исполняемого файла
clean:
	rm -f (OBJS) (TARGET)

# Указываем, что clean и all не являются реальными файлами
.PHONY: all clean
